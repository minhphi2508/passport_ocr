# Deprecated compatibility file.
#
# The maintained NVIDIA GPU configuration is RTX 5060 Ti on Windows.
# It uses TWO isolated GPU virtual environments because PyTorch CUDA and
# PaddlePaddle CUDA DLLs can conflict when loaded in the same Python process.
#
# Do not run:
#   pip install -r requirements-gpu.txt
#
# Instead run:
#
#   Set-ExecutionPolicy -Scope Process -ExecutionPolicy Bypass
#   .\scripts\setup_rtx5060ti.ps1
#
# Then:
#
#   .\scripts\run_rtx5060ti.ps1 --fresh
#
# See README.md and requirements-rtx5060ti.txt.
